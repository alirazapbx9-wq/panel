def get_users_data():
    users = {}
    conf_files = glob.glob(f"{USER_DIR}/*.conf")
    for conf in conf_files:
        uname = os.path.basename(conf).replace('.conf', '')
        ip_limit, gb_limit, used_mb = "1", "1", 0.0
        try:
            with open(conf, 'r') as f:
                for line in f:
                    if line.startswith("IP_LIMIT="): ip_limit = line.strip().split("=")[1]
                    elif line.startswith("GB_LIMIT="): gb_limit = line.strip().split("=")[1]
                    elif line.startswith("USED_MB="): used_mb = float(line.strip().split("=")[1])
        except Exception: pass

        chage_out = run_cmd(f"chage -l {uname}")
        expiry = "N/A"
        for line in chage_out.splitlines():
            if "Account expires" in line:
                expiry = line.split(":")[1].strip()
                break

        passwd_out = run_cmd(f"passwd -S {uname}")
        status = "Active"
        if " L " in f" {passwd_out} ": status = "LOCKED"

        if used_mb < 1024:
            used_str = f"{used_mb:.2f} MB"
        else:
            used_str = f"{(used_mb / 1024.0):.2f} GB"

        users[uname] = {
            "ip_limit": ip_limit,
            "gb_limit": gb_limit,
            "used_str": used_str,
            "expiry": expiry,
            "status": status
        }
    return users

@app.route('/api/server-message', methods=['GET', 'POST'])
def api_server_message():
    if not session.get('logged_in'):
        return jsonify({"error": "Unauthorized"}), 401
    banner_path = "/etc/issue.net"
    if request.method == 'GET':
        try:
            with open(banner_path, "r", encoding="utf-8", errors="replace") as f:
                message = f.read()
        except FileNotFoundError:
            message = ""
        except Exception:
            return jsonify({"error": "Could not read server message"}), 500
        return jsonify({"message": message})
    data = request.get_json(silent=True) or {}
    message = data.get('message', '')
    if not isinstance(message, str):
        return jsonify({"success": False, "message": "Message must be text"}), 400
    if len(message) > 4000:
        return jsonify({"success": False, "message": "Message is too long (max 4000 characters)"}), 400
    try:
        with open(banner_path, "w", encoding="utf-8") as f:
            f.write(message)
        return jsonify({"success": True, "message": "Server message saved successfully!"})
    except Exception:
        return jsonify({"success": False, "message": "Could not save server message"}), 500

@app.route('/api/stats', methods=['GET', 'POST'])
def api_stats():
    if not session.get('logged_in'): return jsonify({"error": "Unauthorized"}), 401
    ram = run_cmd("free -m | awk 'NR==2{printf \"%.1f\", $3*100/$2 }'")
    cpu = run_cmd("top -bn1 | grep 'Cpu(s)' | awk '{print $2 + $4}'")
    
    active_sessions = []
    seen_users = set()
    
    raw_logs = run_cmd("journalctl -u dropbear --no-pager -n 400 2>/dev/null")
    if os.path.exists("/var/log/auth.log"):
        try:
            with open("/var/log/auth.log", "r") as f:
                raw_logs += "\n" + f.read()
        except Exception: pass

    external_ips = run_cmd("ss -tnp 2>/dev/null | grep -E ':(80|443|2082)' | grep 'ESTAB' | awk '{print $5}' | cut -d: -f1 | grep -vE '^127\.|^::1' | sort -u").splitlines()
    ws_logged_ips = run_cmd("grep -oP '(?<=REAL_IP:)\\S+' /var/log/ws-proxy.log 2>/dev/null | tail -n 20 | sort -u").splitlines()
    
    real_ip_pool = list(dict.fromkeys([ip for ip in (external_ips + ws_logged_ips) if ip]))
    ip_index = 0

    pids_out = run_cmd("ps aux | grep dropbear | grep -v grep | awk '{print $2}'").splitlines()
    
    for pid in pids_out:
        if not pid: continue
        user_matches = [l for l in raw_logs.splitlines() if f"dropbear[{pid}]" in l and "Password auth succeeded" in l]
        if user_matches:
            last_line = user_matches[-1]
            m = re.search(r"for \x27(\w+)\x27", last_line)
            if not m:
                m = re.search(r"for (\w+)", last_line)
            
            if m:
                uname = m.group(1)
                if uname in seen_users:
                    continue
                
                seen_users.add(uname)
                ip_match = re.search(r"from (\S+?):", last_line)
                logged_ip = ip_match.group(1) if ip_match else ""
                
                final_ip = logged_ip
                if not logged_ip or logged_ip == "127.0.0.1":
                    if ip_index < len(real_ip_pool):
                        final_ip = real_ip_pool[ip_index]
                        ip_index += 1
                    else:
                        final_ip = "104.16.79.73"

                active_sessions.append({"user": uname, "ip": final_ip})

    return jsonify({"domain": get_domain(), "ram": ram or "0", "cpu": cpu or "0", "users": get_users_data(), "active_sessions": active_sessions})

    return jsonify({"success": False, "message": "User config not found!"})

@app.route('/api/edit-user', methods=['GET', 'POST'])
def edit_user():
    if not session.get('logged_in'): return jsonify({"error": "Unauthorized"}), 401
    d = request.json or {}
    old_u = d.get('old_username','').strip()
    new_u = d.get('new_username','').strip()
    passw = d.get('password','').strip()
    ip_lim = d.get('ip_limit','1').strip()
    gb_lim = d.get('gb_limit','1').strip()
    exp_date = d.get('expiry_date','').strip()

    if not old_u or not new_u:
        return jsonify({"success": False, "message": "Invalid Username!"})

    if old_u != new_u:
        run_cmd(f"pkill -9 -u {old_u}")
        run_cmd(f"usermod -l {new_u} {old_u}")
        if os.path.exists(f"{USER_DIR}/{old_u}.conf"):
            os.rename(f"{USER_DIR}/{old_u}.conf", f"{USER_DIR}/{new_u}.conf")

    target_u = new_u

    if passw:
        run_cmd(f"echo '{target_u}:{passw}' | chpasswd")
        run_cmd(f"passwd -u {target_u} 2>/dev/null")

    if exp_date:
        run_cmd(f"usermod -e '{exp_date}' {target_u}")

    conf_path = f"{USER_DIR}/{target_u}.conf"
    used_mb = 0.0
    if os.path.exists(conf_path):
        with open(conf_path, "r") as f:
            for line in f:
                if line.startswith("USED_MB="):
                    try: used_mb = float(line.strip().split("=")[1])
                    except: pass

    with open(conf_path, "w") as f:
        f.write(f"IP_LIMIT={ip_lim}\nGB_LIMIT={gb_lim}\nUSED_MB={used_mb:.4f}\n")

    return jsonify({"success": True, "message": f"Account '{target_u}' Updated Successfully!"})

# 5. Deploy Optimized Bandwidth Tracker (autokill.py)
echo -e "${BLUE}[3/7] Deploying Bandwidth Tracker...${NC}"
cat << 'PY_EOF' > /usr/local/bin/autokill.py
import os, sys, time, subprocess, re

USER_DIR = "/etc/raretriccks/users"

def get_auth_logs():
    raw = ""
    try:
        raw = subprocess.check_output(["journalctl", "-u", "dropbear", "--no-pager", "-n", "100"], stderr=subprocess.DEVNULL).decode("utf-8", errors="ignore")
    except Exception: pass
    return raw

def get_active_users_and_pids(raw_logs):
    user_pids = {}
    try:
        ps_out = subprocess.check_output(["ps", "aux"], stderr=subprocess.DEVNULL).decode("utf-8", errors="ignore")
        for line in ps_out.splitlines():
            if "dropbear" in line and "grep" not in line:
                parts = line.split()
                if len(parts) > 1:
                    pid = parts[1]
                    matches = [l for l in raw_logs.splitlines() if f"dropbear[{pid}]" in l and "Password auth succeeded" in l]
                    if matches:
                        last_line = matches[-1]
                        m = re.search(r"for \x27(\w+)\x27", last_line)
                        if not m: m = re.search(r"for (\w+)", last_line)
                        if m:
                            uname = m.group(1)
                            if uname not in user_pids: user_pids[uname] = []
                            user_pids[uname].append(pid)
    except Exception: pass
    return user_pids

def get_pid_io_bytes(pid):
    io_file = f"/proc/{pid}/io"
    total_bytes = 0
    if os.path.exists(io_file):
        try:
            with open(io_file, "r") as f:
                for line in f:
                    if line.startswith("rchar:") or line.startswith("wchar:"):
                        total_bytes += int(line.split(":")[1].strip())
        except Exception: pass
    return total_bytes

last_pid_bytes = {}

while True:
    try:
        raw_logs = get_auth_logs()
        user_pids_map = get_active_users_and_pids(raw_logs)

        if os.path.exists(USER_DIR):
            for fname in os.listdir(USER_DIR):
                if not fname.endswith(".conf"): continue
                uname = fname[:-5]
                conf_path = os.path.join(USER_DIR, fname)
                
                ip_limit = 1
                gb_limit = "Unlimited"
                used_mb = 0.0

                with open(conf_path, "r") as f: lines = f.readlines()
                for line in lines:
                    if line.startswith("IP_LIMIT="):
                        try: ip_limit = int(line.strip().split("=")[1])
                        except Exception: pass
                    elif line.startswith("GB_LIMIT="): gb_limit = line.strip().split("=")[1]
                    elif line.startswith("USED_MB="):
                        try: used_mb = float(line.strip().split("=")[1])
                        except Exception: pass

                active_pids = user_pids_map.get(uname, [])

                delta_mb = 0.0
                for pid in active_pids:
                    current_b = get_pid_io_bytes(pid)
                    if pid in last_pid_bytes:
                        diff = current_b - last_pid_bytes[pid]
                        if diff > 0:
                            delta_mb += (diff / (1024.0 * 1024.0))
                    last_pid_bytes[pid] = current_b

                if delta_mb > 0:
                    used_mb += delta_mb
                    new_lines = []
                    for line in lines:
                        if line.startswith("USED_MB="):
                            new_lines.append(f"USED_MB={used_mb:.4f}\n")
                        else:
                            new_lines.append(line)
                    with open(conf_path, "w") as f: f.writelines(new_lines)

                if gb_limit != "Unlimited":
                    try:
                        max_mb = float(gb_limit) * 1024.0
                        if used_mb >= max_mb:
                            subprocess.call(["passwd", "-l", uname], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
                            for pid in active_pids:
                                subprocess.call(["kill", "-9", pid], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
                    except Exception: pass

                if ip_limit > 0 and len(active_pids) > ip_limit:
                    subprocess.call(["passwd", "-l", uname], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
                    for pid in active_pids:
                        subprocess.call(["kill", "-9", pid], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)

    except Exception: pass
    time.sleep(5)

# 6. Deploy Ultra-Fast WebSocket Proxy (ws-proxy.py)
echo -e "${BLUE}[4/7] Deploying Fast WebSocket Proxy Engine...${NC}"
cat << 'WS_EOF' > /usr/local/bin/ws-proxy.py
import socket, threading, select, time

PORT = 2082
TARGET_HOST = '127.0.0.1'
TARGET_PORT = 109
LOG_FILE = '/var/log/ws-proxy.log'

def log_client_ip(ip):
    try:
        with open(LOG_FILE, 'a') as f:
            f.write(f"{time.strftime('%Y-%m-%d %H:%M:%S')} - REAL_IP:{ip}\n")
    except Exception: pass

def handle_client(client_socket, client_addr):
    real_ip = client_addr[0]
    try:
        client_socket.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
        client_socket.settimeout(10)
        request = client_socket.recv(4096).decode('utf-8', errors='ignore')
        if not request:
            client_socket.close()
            return

        for line in request.split('\r\n'):
            line_l = line.lower()
            if line_l.startswith('cf-connecting-ip:'):
                real_ip = line.split(':')[1].strip()
                break
            elif line_l.startswith('x-forwarded-for:'):
                parts = line.split(':')
                if len(parts) > 1: real_ip = parts[1].strip().split(',')[0].strip()
            elif line_l.startswith('x-real-ip:'):
                parts = line.split(':')
                if len(parts) > 1: real_ip = parts[1].strip().split(',')[0].strip()

        log_client_ip(real_ip)

        response = "HTTP/1.1 101 Switching Protocols\r\nUpgrade: websocket\r\nConnection: Upgrade\r\n\r\n"
        client_socket.sendall(response.encode('utf-8'))

        target_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        target_socket.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
        target_socket.connect((TARGET_HOST, TARGET_PORT))

        sockets = [client_socket, target_socket]
        client_socket.settimeout(None)

        while True:
            readable, _, _ = select.select(sockets, [], [], 30)
            if not readable: break
            for s in readable:
                other = target_socket if s is client_socket else client_socket
                data = s.recv(16384)
                if not data:
                    client_socket.close()
                    target_socket.close()
                    return
                other.sendall(data)
    except Exception: pass
    finally:
        try: client_socket.close()
        except: pass
        try: target_socket.close()
        except: pass

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(('0.0.0.0', PORT))
server.listen(500)

while True:
    client, addr = server.accept()
    threading.Thread(target=handle_client, args=(client, addr), daemon=True).start()
WS_EOF
